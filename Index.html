Add the Apps Script

In the spreadsheet, click Extensions → Apps Script. Delete all existing code and paste the code below exactly as it is.

Apps Script — paste this into the editor in Step 2 
const SHEET_NAME = 'Votes';

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    if (data.action === 'reset') {
      const ss = SpreadsheetApp.getActiveSpreadsheet();
      const sheet = ss.getSheetByName(SHEET_NAME);
      if (sheet && sheet.getLastRow() > 1) {
        sheet.deleteRows(2, sheet.getLastRow() - 1);
      }
      return ContentService
        .createTextOutput(JSON.stringify({ ok: true, action: 'reset' }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      sheet = ss.insertSheet(SHEET_NAME);
      sheet.appendRow(['timestamp', 'votes_json']);
    }
    sheet.appendRow([new Date().toISOString(), JSON.stringify(data.votes)]);
    return ContentService
      .createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet || sheet.getLastRow() < 2) {
      return ContentService
        .createTextOutput(JSON.stringify({ ok: true, votes: [] }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    const rows = sheet.getRange(2, 2, sheet.getLastRow() - 1, 1).getValues();
    const votes = rows.map(r => JSON.parse(r[0]));
    return ContentService
      .createTextOutput(JSON.stringify({ ok: true, votes: votes }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
